cmake_minimum_required(VERSION 3.20)
project(book02_numerical_geometry LANGUAGES CXX)
add_executable(book02_workbench src/main.cpp)
target_compile_features(book02_workbench PRIVATE cxx_std_23)
set_target_properties(book02_workbench PROPERTIES CXX_EXTENSIONS OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(book02_workbench PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_test(NAME book02_cpp23_mathematical_contracts COMMAND book02_workbench test)
  add_test(NAME book02_cpp23_heat_export COMMAND book02_workbench heat)
  add_test(NAME book02_cpp23_sampling_export COMMAND book02_workbench monte-carlo)
  add_test(NAME book02_cpp23_qr_export COMMAND book02_workbench qr-sensitivity)
  add_test(NAME book02_cpp23_robust_summary COMMAND book02_workbench robust-fit)
  add_test(NAME book02_cpp23_robust_data_export COMMAND book02_workbench robust-data)
  add_test(NAME book02_cpp23_ransac_summary COMMAND book02_workbench ransac-summary)
  add_test(NAME book02_cpp23_ransac_data_export COMMAND book02_workbench ransac-data)
  add_test(NAME book02_cpp23_exact_orientation_export COMMAND book02_workbench orientation-exact)
  add_test(NAME book02_cpp23_exact_orientation_audit_export COMMAND book02_workbench orientation-audit)
  find_package(Python3 QUIET COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME book02_cpp23_binary64_independent_rational_oracle
      COMMAND "${Python3_EXECUTABLE}"
              "${CMAKE_CURRENT_SOURCE_DIR}/predicate_oracle.py"
              "$<TARGET_FILE:book02_workbench>")
  endif()
endif()
